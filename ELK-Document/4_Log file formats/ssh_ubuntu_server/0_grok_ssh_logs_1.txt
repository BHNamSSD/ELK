Dùng aggregate filter trong Logstash để gom nhiều log (cùng pid của sshd) → thành 1 event duy nhất.

----------Cách hoạt động
Mỗi kết nối SSH đều gắn với 1 PID (vd: sshd[10857]).
Mình lấy pid làm “chìa khóa” (task_id).
Logstash sẽ gom tất cả log có cùng pid vào 1 nhóm.
Sau vài giây (timeout), nếu không còn log mới từ pid đó → Logstash sẽ xuất ra 1 event duy nhất.

filter {
  grok {
	#collent logs
    match => {
      "message" => "%{TIMESTAMP_ISO8601:timestamp} %{HOSTNAME:hostname} sshd\[%{NUMBER:pid}\]: %{GREEDYDATA:ssh_msg}"
    }
  }

  aggregate {
    task_id => "%{pid}"          # gom theo PID
    code => "
      map['timestamp'] ||= event.get('timestamp')
      map['hostname']  ||= event.get('hostname')
      map['pid']       ||= event.get('pid')

      # lưu tất cả các log vào mảng
      map['details'] ||= []
      map['details'] << event.get('ssh_msg')





      # nếu có dòng Failed password thì trích user, ip, port
      if event.get('ssh_msg') =~ /Failed password for/
        map['user'] = event.get('ssh_msg').match(/Failed password for (\w+)/)[1] rescue nil
        map['src_ip'] = event.get('ssh_msg').match(/from ([0-9.]+)/)[1] rescue nil
        map['src_port'] = event.get('ssh_msg').match(/port (\d+)/)[1] rescue nil
        map['status'] = 'failed'
      end
	  
	      
	  
	  # Successful login
      if event.get('ssh_msg') =~ /Accepted password for/
        map['user']    = event.get('ssh_msg').match(/Accepted password for (\w+)/)[1] rescue nil
        map['src_ip']  = event.get('ssh_msg').match(/from ([0-9.]+)/)[1] rescue nil
        map['src_port']= event.get('ssh_msg').match(/port (\d+)/)[1] rescue nil
        map['status']  = 'success'
        map['auth_method'] = 'password'
      end
	


	
	  # Session opened
      if event.get('ssh_msg') =~ /session opened for user/
        map['session_status'] = 'opened'
      end
	  
    "
	
	
    push_map_as_event_on_timeout => true
    timeout_task_id_field => "pid"
    timeout => 5
  }
}

filter {
  grok {
    match => {
      "message" => "%{TIMESTAMP_ISO8601:timestamp} %{HOSTNAME:hostname} %{SYSLOGPROG:program}\[%{NUMBER:pid}\]: %{GREEDYDATA:ssh_msg}"
    }
    tag_on_failure => ["_grokfail_ssh"]
  }

  if [program] == "sshd" {
    aggregate {
      task_id => "%{pid}"
      code => "
        map['timestamp'] ||= event.get('timestamp')
        map['hostname'] ||= event.get('hostname')
        map['pid'] ||= event.get('pid')

        map['details'] ||= []
        map['details'] << event.get('ssh_msg')

        # -------------------
        # LOGIN FAILED
        if event.get('ssh_msg') =~ /Failed password for/
          map['user'] = event.get('ssh_msg').match(/Failed password for (?:invalid user )?(\S+)/)[1] rescue nil
          map['src_ip'] = event.get('ssh_msg').match(/from ([0-9.]+)/)[1] rescue nil
          map['src_port'] = event.get('ssh_msg').match(/port (\d+)/)[1] rescue nil
          map['status'] = 'failed'
        end

        # LOGIN SUCCESS
        if event.get('ssh_msg') =~ /Accepted password for/
          map['user'] = event.get('ssh_msg').match(/Accepted password for (\S+)/)[1] rescue nil
          map['src_ip'] = event.get('ssh_msg').match(/from ([0-9.]+)/)[1] rescue nil
          map['src_port'] = event.get('ssh_msg').match(/port (\d+)/)[1] rescue nil
          map['status'] = 'success'
          map['auth_method'] = 'password'
        end

        # INVALID USER
        if event.get('ssh_msg') =~ /Invalid user/
          map['user'] = event.get('ssh_msg').match(/Invalid user (\S+)/)[1] rescue nil
          map['src_ip'] = event.get('ssh_msg').match(/from ([0-9.]+)/)[1] rescue nil
          map['src_port'] = event.get('ssh_msg').match(/port (\d+)/)[1] rescue nil
          map['status'] = 'invalid_user'
        end

        # DISCONNECT
        if event.get('ssh_msg') =~ /Received disconnect/
          map['src_ip'] ||= event.get('ssh_msg').match(/from ([0-9.]+)/)[1] rescue nil
          map['src_port'] ||= event.get('ssh_msg').match(/port (\d+)/)[1] rescue nil
          map['status'] = 'disconnect'
          map['reason'] = event.get('ssh_msg').match(/: (.*?)(?: \\[preauth\\])?$/)[1] rescue nil
        end

        if event.get('ssh_msg') =~ /Disconnected from/
          map['src_ip'] ||= event.get('ssh_msg').match(/ ([0-9.]+)/)[1] rescue nil
          map['src_port'] ||= event.get('ssh_msg').match(/port (\d+)/)[1] rescue nil
          map['status'] ||= 'disconnected'
        end

        # SESSION OPENED/CLOSED
        if event.get('ssh_msg') =~ /session opened/
          map['session'] = 'opened'
        end
        if event.get('ssh_msg') =~ /session closed/
          map['session'] = 'closed'
        end
      "
      push_map_as_event_on_timeout => true
      timeout_task_id_field => "pid"
      timeout => 20
    }
  }
}
